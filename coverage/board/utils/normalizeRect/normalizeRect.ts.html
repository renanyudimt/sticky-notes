import type { Position, Rect } from '@/modules/notes';
 
/**
 * Build a positive-sized rectangle from two corner points, regardless of drag
 * direction (so dragging up-and-left still yields a valid rect).
 */
export function normalizeRect(start: Position, end: Position): Rect {
  return {
    x: Math.min(start.x, end.x),
    y: Math.min(start.y, end.y),
    width: Math.abs(end.x - start.x),
    height: Math.abs(end.y - start.y),
  };
}
